# Serving stack for the DeepSeek-V4-Pro recipe on the CUDA 13.0 H200 image.

# vLLM's CUDA requirements for the current release line. vLLM itself pins these
# base versions; the local suffix additionally locks every wheel to CUDA 13.0.
torch==2.13.0+cu130
torchvision==0.28.0+cu130
torchaudio==2.11.0+cu130

# FlashInfer compiles SM90 kernels during vLLM warmup and invokes this binary.

# Client for smoke_test.py and bench.sh.

# prepare_model.py / download.py. vLLM happens to pull huggingface_hub in
# transitively; declaring it keeps that from being luck. hf_xet is what gives

huggingface_hub[hf_xet]>=1.0.0
